        };
        btnHomeEmptyCreateSession.setOnClickListener(createSessionListener);

        tvHomeViewHistory.setOnClickListener(v -> {
            openHistory();
        });

    }

    private void refreshHomeData() {
        if(AttendanceRepository.getInstance().isAdmin())return;
        int suggested=homeSuggestions==null?0:homeSuggestions.render(System.currentTimeMillis());
        AttendanceSession active = AttendanceRepository.getInstance().getActiveSession();
        if (active != null) {
            cardActiveSession.setVisibility(View.VISIBLE);
            cardNoActiveSession.setVisibility(View.GONE);

            tvHomeActiveTitle.setText(active.getTitle());
            ClassRoom cr = AttendanceRepository.getInstance().getClassById(active.getClassId());
            if (cr != null) {
                tvHomeActiveClass.setText(AttendanceRepository.getInstance().sessionClassLabel(active));
            }
            tvHomeActiveTime.setText(active.getStartTime() + " • " + active.getDate()+"\n"+active.getTeacherNameSnapshot());

            int present = AttendanceRepository.getInstance().getPresentCount(active.getId());
            int late = AttendanceRepository.getInstance().getLateCount(active.getId());
            int unchecked = AttendanceRepository.getInstance().getUncheckedCount(active.getId());
            int checkedIn = present + late;
            int total = active.getExpectedCount();
            int pct = total > 0 ? (checkedIn * 100 / total) : 0;

            tvHomeProgressLabel.setText(checkedIn + "/" + total + " đã điểm danh (" + pct + "%)");
            tvHomeBreakdownLabel.setText(present + " Có mặt • " + late + " Đi muộn • " + unchecked + " Chưa điểm danh");
            progressHomeActive.setProgress(pct);
        } else {
            cardActiveSession.setVisibility(View.GONE);
            cardNoActiveSession.setVisibility(suggested>0?View.GONE:View.VISIBLE);
        }
        btnHomeEmptyCreateSession.setText(AttendanceRepository.getInstance().getAllClasses().isEmpty()?(AttendanceRepository.getInstance().isAdmin()?"Tạo lớp":"Chưa được phân công lớp"):"Tạo buổi ngoài lịch");

        // Recent sessions (finished sessions)
        List<AttendanceSession> all = AttendanceRepository.getInstance().getAllSessions();
        List<AttendanceSession> recent = new ArrayList<>();
        for (AttendanceSession s : all) {
            if (s.getStatus() == AttendanceSession.SessionStatus.CLOSED) {
                recent.add(s);
                if (recent.size() >= 2) break;
            }
        }
        homeRecentAdapter.setSessions(recent);
    }

    private void setupClassesTab() {
        classAdapter = new ClassAdapter(this, classRoom -> {

            intent.putExtra(ClassDetailActivity.EXTRA_CLASS_ID, classRoom.getId());

        };

        btnClassesEmptyAdd.setOnClickListener(addClassListener);

        etClassesSearch.addTextChangedListener(new TextWatcher() {
            @Override public void beforeTextChanged(CharSequence s, int start, int count, int after) {}
            @Override public void onTextChanged(CharSequence s, int start, int before, int count) {
                refreshClassesData(s.toString());
            }
            @Override public void afterTextChanged(Editable s) {}
        });

        refreshClassesData("");
    }

    private void refreshClassesData(String query) {
        List<ClassRoom> all = AttendanceRepository.getInstance().getAllClasses();
        if (all.isEmpty()) {
            llClassesEmpty.setVisibility(View.VISIBLE);
            rvClasses.setVisibility(View.GONE);
            tvClassesNoSearch.setVisibility(View.GONE);
            return;
        }

        llClassesEmpty.setVisibility(View.GONE);
        List<ClassRoom> filtered = new ArrayList<>();
        String q = query != null ? query.trim().toLowerCase() : "";

        for (ClassRoom c : all) {
            if (q.isEmpty() || c.getName().toLowerCase().contains(q) || c.getCode().toLowerCase().contains(q)) {
                filtered.add(c);
            }
        }

        if (filtered.isEmpty()) {
            tvClassesNoSearch.setVisibility(View.VISIBLE);
            rvClasses.setVisibility(View.GONE);
        } else {
            tvClassesNoSearch.setVisibility(View.GONE);
            rvClasses.setVisibility(View.VISIBLE);
            filtered.sort((a,b)->classSortDescending?b.getName().compareToIgnoreCase(a.getName()):a.getName().compareToIgnoreCase(b.getName()));
            classAdapter.setClasses(filtered);
        }
    }

    // --- 3. HISTORY TAB SETUP ---
    private void setupHistoryTab() {
        historyAdapter = new SessionAdapter(this, session -> {
            Intent intent = new Intent(this, SessionDetailActivity.class);
            intent.putExtra(SessionDetailActivity.EXTRA_SESSION_ID, session.getId());
            startActivity(intent);
        });
        rvHistory.setLayoutManager(new LinearLayoutManager(this));
        rvHistory.setAdapter(historyAdapter);

        cardHistoryStatsBanner.setOnClickListener(v -> {
            Intent intent = new Intent(this, AttendanceStatsActivity.class);
            Bundle selection=new Bundle();filters.save(selection);intent.putExtra("filters",selection);
            startActivity(intent);
        });

        filters.classes(chipGroupHistoryClass);

        btnHistoryResetFilter.setOnClickListener(v -> {
            filters.classId=null;filters.from=null;filters.until=null;filters.status=null;filters.classes(chipGroupHistoryClass);
            refreshHistoryData();
        });

        refreshHistoryData();
    }

    private void refreshHistoryData() {
        List<AttendanceSession> all = AttendanceRepository.getInstance().getAllSessions();

        List<AttendanceSession> filtered = new ArrayList<>();
        for (AttendanceSession s : all) {
            if (filters==null||filters.matches(s)) {
                filtered.add(s);
            }
        }

        if (filtered.isEmpty()) {
            llHistoryEmptyFilter.setVisibility(View.VISIBLE);
            rvHistory.setVisibility(View.GONE);
        } else {
            llHistoryEmptyFilter.setVisibility(View.GONE);
            rvHistory.setVisibility(View.VISIBLE);
            historyAdapter.setSessions(filtered);
        }
    }
}

package com.example.th9.data;

import android.content.Context;
import android.os.Looper;
import androidx.room.Room;
import com.example.th9.model.*;
import com.example.th9.data.AttendanceDatabase.*;
import java.text.*;
import java.util.*;
import java.util.concurrent.Callable;
import static com.example.th9.data.AttendanceRules.require;

/** Room is authoritative. UI reads detached display models from an atomically published snapshot.
 * All disk access and mutations run on Tasks.IO, including startup and process restoration. */
public final class AttendanceRepository {
    private static AttendanceRepository instance;
    private final AttendanceDatabase db;
    private final Store dao;
    private final AuthStore auth;
    private volatile Snapshot snapshot = new Snapshot();
    private boolean initialized;
    private static class Snapshot {
        List<ClassSchedule> schedules=new ArrayList<>();
        List<UserAccount> users=new ArrayList<>();
        List<TeacherClass> assignments=new ArrayList<>();
        List<Classroom> classes = new ArrayList<>();
        List<StudentRow> students = new ArrayList<>();
        List<ClassMembership> members = new ArrayList<>();
        List<SessionRow> sessions = new ArrayList<>();
        List<RecordRow> records = new ArrayList<>();
    }
    public static synchronized void configure(Context context) {
        if (instance == null) instance = new AttendanceRepository(Room.databaseBuilder(context,
                AttendanceDatabase.class,"attendance.db").addMigrations(AttendanceDatabase.MIGRATION_1_2,AttendanceDatabase.MIGRATION_2_3,AttendanceDatabase.MIGRATION_3_4,AttendanceDatabase.MIGRATION_4_5).build(),context.getSharedPreferences("auth",Context.MODE_PRIVATE));
    }
    public static AttendanceRepository getInstance() {
        if (instance == null) throw new IllegalStateException("Chưa khởi tạo dữ liệu");
        return instance;
    }
    public AttendanceRepository(AttendanceDatabase database){this(database,null);}
    public AttendanceRepository(AttendanceDatabase database,android.content.SharedPreferences preferences){db=database;dao=db.store();auth=new AuthStore(dao,preferences);}
    private void worker() { require(Looper.myLooper()!=Looper.getMainLooper(), "Phải truy cập Room ngoài main thread"); }
    public synchronized void initialize() { worker(); if (!initialized) { reload(); initialized=true; } }
    public synchronized boolean isInitialized(){return initialized;}
    public synchronized void reload() {
        worker();
        snapshot=db.runInTransaction(() -> {
            Snapshot s=new Snapshot(); s.schedules=dao.schedules(); s.classes=dao.classes(); s.students=dao.students();
            s.members=dao.memberships(); s.sessions=dao.sessions(); s.records=dao.records();s.users=dao.users();s.assignments=dao.assignments();auth.refresh(); return s;
        });
    }
    private synchronized <T> T write(Callable<T> action) {
        worker();
        T result=db.runInTransaction(action);
        reload();
        return result;
    }
    private <T> T writeAdmin(Callable<T> action){return write(()->{auth.admin();return action.call();});}
    public Account currentAccount(){return auth.current();}
    public boolean isAdmin(){Account user=currentAccount();return user!=null&&user.isAdmin();}
    public boolean needsAdminSetup(){return snapshot.users.isEmpty();}
    public void setupAdmin(String username,String name,String password){write(()->{auth.setup(username,name,password);return null;});}

    public void replaceClassTeachers(String classId,List<String> teacherIds){
        Set<String> desired=new HashSet<>(teacherIds);
        writeAdmin(()->{
            require(dao.classroom(classId)!=null,"Lớp không tồn tại");
            require(desired.size()<=1,"Mỗi lớp chỉ được phân công một giảng viên");
            List<TeacherClass> all=dao.assignments();
            for(String teacherId:desired){
                UserAccount teacher=dao.user(teacherId);require(teacher!=null&&"TEACHER".equals(teacher.role),"Không tìm thấy giảng viên");
                if(dao.assigned(teacherId,classId)==0)require(teacher.enabled,"Tài khoản giảng viên đang bị khóa");
                Set<String> classes=new HashSet<>();for(TeacherClass a:all)if(a.teacherId.equals(teacherId))classes.add(a.classId);
                classes.add(classId);validateAssignmentSchedules(classes);
            }
            for(SessionRow session:dao.sessions())if(classId.equals(session.classId)&&"OPEN".equals(session.status)&&session.teacherId!=null&&dao.assigned(session.teacherId,classId)>0)
                require(desired.contains(session.teacherId),"Hãy kết thúc buổi đang mở trước khi hủy phân công");
            for(TeacherClass a:all)if(a.classId.equals(classId)&&!desired.contains(a.teacherId))dao.unassign(a.teacherId,classId);
            for(String teacherId:desired){TeacherClass a=new TeacherClass();a.teacherId=teacherId;a.classId=classId;dao.assign(a);}
            return null;
        });
    }

    public static long parseDateTime(String date,String time) {
        try {
            SimpleDateFormat f=new SimpleDateFormat("dd/MM/yyyy HH:mm",Locale.getDefault()); f.setLenient(false);
            String text=date+" "+time;
            ParsePosition position=new ParsePosition(0); Date parsed=f.parse(text,position);
            require(parsed!=null && position.getIndex()==text.length(),"Ngày giờ không hợp lệ");
            return parsed.getTime();
        } catch (RuntimeException e) { throw new IllegalArgumentException("Ngày giờ không hợp lệ"); }
    }

    private AttendanceSession sessionModel(SessionRow r,Snapshot s) {
        int expected=0;
        if(r.status.equals("DRAFT")) { for(ClassMembership m:s.members) if(m.classId.equals(r.classId)) expected++; }
        else for(RecordRow rec:s.records) if(rec.sessionId.equals(r.id)) expected++;
        AttendanceSession model=new AttendanceSession(r.id,r.classId,r.title,format(r.scheduledAt,"dd/MM/yyyy"),
                format(r.scheduledAt,"HH:mm"),r.lateAt!=null,format(r.lateAt,"HH:mm"),
                AttendanceSession.SessionStatus.valueOf(r.status),r.closedAt==null?null:format(r.closedAt,"HH:mm, dd/MM/yyyy"),expected);
        model.setSnapshot(r.classNameSnapshot,r.classCodeSnapshot,r.scheduledAt,r.openedAt,r.closedAt);
        model.setTeacher(r.teacherId,r.teacherNameSnapshot,r.teacherUsernameSnapshot);
        model.setSchedule(r.scheduleId,r.plannedDate,r.plannedShift,r.plannedStartMinute,r.plannedEndMinute);

    /** Allocates a cohort number and creates the profile/card in the same transaction.
     * A stable creation ID makes a retry after Activity recreation return the saved profile. */
    public Student registerStudent(Student model,String classId) {
        return writeAdmin(()->{
            StudentRow existing=model.getId()==null?null:dao.student(model.getId());
            if(existing!=null){require(!existing.archived,"Hồ sơ đã được lưu trữ");return studentModel(existing);}
            ProfileRules.validateRegistration(model.getFullName(),model.getBirthDate(),model.getCourse(),model.getGender());
            String course=ProfileRules.normalizeCourse(model.getCourse());
            StudentRow row=studentRow(model);row.course=course;
            row.mssv=allocateStudentNumber(course);row.digitalCardCode=row.mssv;row.cardCode=null;
            if(classId!=null){require(dao.classroom(classId)!=null,"Lớp không tồn tại");row.primaryClassId=classId;}
            validateStudent(row);dao.insertStudent(row);
            if(classId!=null)member(classId,row.id);
            model.setId(row.id);model.setMssv(row.mssv);return studentModel(row);
        });
    }
    private String allocateStudentNumber(String course){
        String prefix=course.substring(0,4)+course.substring(5,9);
        StudentNumberSequence sequence=dao.numberSequence(prefix);
        if(sequence==null){sequence=new StudentNumberSequence();sequence.prefix=prefix;}
        // Include archived identities and legacy card codes so an existing code is never reused.
        for(StudentRow other:dao.students())for(String code:new String[]{other.mssv,other.cardCode,other.digitalCardCode}){
            if(code!=null&&code.length()==12&&code.startsWith(prefix)&&code.substring(8).matches("[0-9]{4}"))
                sequence.lastNumber=Math.max(sequence.lastNumber,Integer.parseInt(code.substring(8)));
        }
        require(sequence.lastNumber<9999,"Niên khóa này đã cấp đủ 9.999 MSSV");
        sequence.lastNumber++;dao.saveNumberSequence(sequence);
        return prefix+String.format(Locale.ROOT,"%04d",sequence.lastNumber);
    }

            return added;
        });
    }

    public String saveSchedule(ClassSchedule proposed,boolean overwriteExisting) {
        ClassSchedule row=proposed.copy();
        return writeAdmin(()->{
            require(dao.classroom(row.classId)!=null,"Chọn lớp học");
            ClassSchedule old=row.id.isEmpty()?null:dao.schedule(row.id);
            require(row.id.isEmpty()||old!=null,"Lịch không còn tồn tại");
            require(old==null||(!old.archived&&old.classId.equals(row.classId)),"Không đổi lớp hoặc sửa lịch đã lưu trữ");
            List<ClassSchedule> all=dao.schedules();
            List<ClassSchedule> others=new ArrayList<>();
            for(ClassSchedule other:all)if(!other.archived&&other.classId.equals(row.classId)&&!other.id.equals(row.id))others.add(other);
            require(others.isEmpty()||overwriteExisting,SCHEDULE_EXISTS);
            if(old==null&&!others.isEmpty()){
                old=others.remove(0);row.id=old.id;
            }
            row.archived=false;ScheduleRules.validate(row);
            Set<String> teachers=new HashSet<>();for(TeacherClass a:dao.assignments())if(a.classId.equals(row.classId))teachers.add(a.teacherId);
            require(!teachers.isEmpty(),"Hãy phân công giảng viên cho lớp trước khi xếp lịch");
            for(ClassSchedule other:all)if(!other.classId.equals(row.classId)&&ScheduleRules.overlaps(row,other)){
                for(TeacherClass a:dao.assignments())if(a.classId.equals(other.classId))
                    require(!teachers.contains(a.teacherId),"Giảng viên trùng lịch với lớp "+dao.classroom(other.classId).name);
            }
            for(ClassSchedule duplicate:others){duplicate.archived=true;dao.updateSchedule(duplicate);}
            if(old==null){row.id=id();dao.insertSchedule(row);}else dao.updateSchedule(row);
            return row.id;
        });
    }

                if(error==null && found!=null) {
                    if(!found.fullName.equals(row.name)||(row.hasCardColumn&&!Objects.equals(found.cardCode,row.card)))error="Thông tin không khớp với hồ sơ đã có";
                    else action=classId==null||dao.membership(classId,found.id)>0?"Bỏ qua":"Thêm vào lớp";
                }
                StudentRow newRow=null;
                if(error==null && found==null) {
                    newRow=new StudentRow();newRow.id=id();newRow.mssv=row.mssv;newRow.fullName=row.name;newRow.cardCode=row.card;
                    for(StudentRow other:proposed) {

                    }
                    if(error==null){try{validateStudent(newRow);}catch(IllegalArgumentException e){error=e.getMessage();}}
                    if(error==null){proposed.add(newRow);action=classId==null?"Tạo hồ sơ sinh viên":"Tạo hồ sơ và thêm vào lớp";}
                }
                preview.lines.add("Dòng "+row.line+" · "+row.mssv+" · "+row.name+"\n"+(error==null?action:"Lỗi: "+error));

                preview.students.add(found==null?newRow:found);preview.actions.add(action);
            }
            if(commit) {

package com.example.th9.data;
import android.content.SharedPreferences;
import com.example.th9.data.AttendanceDatabase.*;
import com.example.th9.model.Account;
import java.security.MessageDigest;
import java.nio.charset.StandardCharsets;
import java.util.Locale;
import static com.example.th9.data.AttendanceRules.require;
/** Called within repository transactions; cached identity is for detached UI reads only. */
final class AuthStore {
    private final Store dao;private final SharedPreferences prefs;
    private String token;
    private volatile Account cached;
    private volatile long validUntil;
    AuthStore(Store dao,SharedPreferences prefs){this.dao=dao;this.prefs=prefs;token=prefs==null?"":prefs.getString("session_token","");}
    static Account account(UserAccount u){return new Account(u.id,u.username,u.fullName,u.role,u.enabled);}
    private UserAccount validActor(){
        LocalLogin login=dao.login();if(token.isEmpty()||login==null||login.expiresAt<=System.currentTimeMillis())return null;
        if(!MessageDigest.isEqual(login.tokenHash.getBytes(StandardCharsets.UTF_8),PasswordHasher.tokenHash(token).getBytes(StandardCharsets.UTF_8)))return null;
        UserAccount user=dao.user(login.userId);return user!=null&&user.enabled?user:null;
    }
    void refresh(){UserAccount user=validActor();cached=user==null?null:account(user);LocalLogin login=dao.login();validUntil=login==null?0:login.expiresAt;}
    Account current(){return System.currentTimeMillis()<validUntil?cached:null;}
    Account requireCurrent(){Account user=current();require(user!=null,"Vui lòng đăng nhập lại");return user;}
    UserAccount actor(){UserAccount user=validActor();require(user!=null,"Vui lòng đăng nhập lại");return user;}
    void admin(){require("ADMIN".equals(actor().role),"Chỉ quản trị viên được thực hiện thao tác này");}
    boolean needsSetup(){return dao.users().isEmpty();}
    static String username(String value){String name=value==null?"":value.trim().toLowerCase(Locale.ROOT);require(name.matches("[a-z0-9._-]{3,40}"),"Tên đăng nhập gồm 3–40 ký tự a-z, số, dấu chấm, gạch dưới hoặc gạch ngang");return name;}
    static String fullName(String value){String name=value==null?"":value.trim();require(!name.isEmpty()&&name.length()<=120,"Nhập họ tên (tối đa 120 ký tự)");return name;}
    void setup(String name,String fullName,String password){
        require(needsSetup(),"Tài khoản quản trị đã được tạo");UserAccount user=new UserAccount();user.id=AttendanceRepository.id();user.username=username(name);user.fullName=fullName(fullName);user.role="ADMIN";PasswordHasher.set(user,password);dao.insertUser(user);startSession(user);
    }
    String login(String username,String password){
        UserAccount user=dao.username(username==null?"":username.trim().toLowerCase(Locale.ROOT));long now=System.currentTimeMillis();
        if(user==null||!user.enabled)return "Tên đăng nhập hoặc mật khẩu không đúng, hoặc tài khoản đã bị khóa";
        if(user.lockedUntil>now)return "Tạm khóa đăng nhập. Thử lại sau "+Math.max(1,(user.lockedUntil-now+999)/1000)+" giây";
        if(!PasswordHasher.matches(user,password)){
            user.failedAttempts++;if(user.failedAttempts>=5){user.lockedUntil=now+60000;user.failedAttempts=0;}dao.updateUser(user);return "Tên đăng nhập hoặc mật khẩu không đúng";
        }
        user.failedAttempts=0;user.lockedUntil=0;dao.updateUser(user);startSession(user);return null;
    }
    private void startSession(UserAccount user){
        String next=PasswordHasher.randomToken(32);LocalLogin login=new LocalLogin();login.userId=user.id;login.tokenHash=PasswordHasher.tokenHash(next);login.expiresAt=System.currentTimeMillis()+7L*24*60*60*1000;dao.saveLogin(login);
        if(prefs!=null)require(prefs.edit().putString("session_token",next).commit(),"Không thể lưu phiên đăng nhập");token=next;
    }
    void logout(){dao.clearLogin();token="";cached=null;validUntil=0;if(prefs!=null)prefs.edit().remove("session_token").commit();}
}

package com.example.th9.ui;

import android.os.Bundle;

public class ClassFormActivity extends com.example.th9.ui.DataActivity {

    public static final String EXTRA_CLASS_ID = "extra_class_id";

    private String classId;
    private String creationId;
    private boolean isEditMode = false;
    private boolean isFormDirty = false;

    private Toolbar toolbar;
    private TextInputLayout tilName, tilCode, tilNote;
    private TextInputEditText etName, etCode, etNote;
    private MaterialButton btnSubmit, btnCancel;

    private String originalName = "";
    private String originalCode = "";
    private String originalNote = "";

    @Override
    protected void onReady(Bundle savedInstanceState) {

        setContentView(R.layout.activity_class_form);

        classId = getIntent().getStringExtra(EXTRA_CLASS_ID);
        creationId=savedInstanceState==null?AttendanceRepository.id():savedInstanceState.getString("creation_id",AttendanceRepository.id());
        isEditMode = (classId != null);

        initViews();
        setupToolbar();
        setupData();
        setupListeners();

        getOnBackPressedDispatcher().addCallback(this, new OnBackPressedCallback(true) {
            @Override
            public void handleOnBackPressed() {
                checkDirtyAndExit();
            }
        });
    }

    private void initViews() {
        toolbar = findViewById(R.id.toolbar_class_form);
        tilName = findViewById(R.id.til_class_name);
        tilCode = findViewById(R.id.til_class_code);
        tilNote = findViewById(R.id.til_class_note);
        etName = findViewById(R.id.et_class_name);
        etCode = findViewById(R.id.et_class_code);
        etNote = findViewById(R.id.et_class_note);
        btnSubmit = findViewById(R.id.btn_class_submit);
        btnCancel = findViewById(R.id.btn_class_cancel);
    }

    private void setupToolbar() {
        setSupportActionBar(toolbar);
        if (getSupportActionBar() != null) {
            getSupportActionBar().setDisplayHomeAsUpEnabled(true);
            getSupportActionBar().setTitle(isEditMode ? R.string.class_form_edit_title : R.string.class_form_create_title);
        }
        toolbar.setNavigationOnClickListener(v -> checkDirtyAndExit());
    }

    private void setupData() {
        if (isEditMode) {
            ClassRoom c = AttendanceRepository.getInstance().getClassById(classId);
            if (c != null) {
                originalName = c.getName();
                originalCode = c.getCode();
                originalNote = c.getNote() != null ? c.getNote() : "";

                etName.setText(originalName);
                etCode.setText(originalCode);
                etNote.setText(originalNote);
            }
            btnSubmit.setText("Lưu thay đổi");
        } else {
            btnSubmit.setText("Tạo lớp học");
        }
    }

    private void setupListeners() {
        TextWatcher dirtyWatcher = new TextWatcher() {
            @Override public void beforeTextChanged(CharSequence s, int start, int count, int after) {}
            @Override public void onTextChanged(CharSequence s, int start, int before, int count) {
                checkIfDirty();
                tilName.setError(null);
                tilCode.setError(null);
            }
            @Override public void afterTextChanged(Editable s) {}
        };

        etName.addTextChangedListener(dirtyWatcher);
        etCode.addTextChangedListener(dirtyWatcher);
        etNote.addTextChangedListener(dirtyWatcher);

        btnSubmit.setOnClickListener(v -> validateAndSubmit());
        btnCancel.setOnClickListener(v -> checkDirtyAndExit());
    }

    private void checkIfDirty() {
        String name = etName.getText() != null ? etName.getText().toString().trim() : "";
        String code = etCode.getText() != null ? etCode.getText().toString().trim() : "";
        String note = etNote.getText() != null ? etNote.getText().toString().trim() : "";

        if (isEditMode) {
            isFormDirty = !name.equals(originalName) || !code.equals(originalCode) || !note.equals(originalNote);
        } else {
            isFormDirty = !name.isEmpty() || !code.isEmpty() || !note.isEmpty();
        }
    }

    private void validateAndSubmit() {
        String name = etName.getText() != null ? etName.getText().toString().trim() : "";
        String code = etCode.getText() != null ? etCode.getText().toString().trim() : "";
        String note = etNote.getText() != null ? etNote.getText().toString().trim() : "";

        boolean hasError = false;
        if (name.isEmpty()) {
            tilName.setError(getString(R.string.class_form_error_name_required));
            hasError = true;
        } else {
            tilName.setError(null);
        }

        if (hasError) return;
        com.example.th9.data.Tasks.run(this,btnSubmit,()->{
            ClassRoom c=new ClassRoom(isEditMode?classId:creationId,name,code,note);
            if(isEditMode)AttendanceRepository.getInstance().updateClass(c);
            else AttendanceRepository.getInstance().addClass(c);
            return c;
        },c->{
            Toast.makeText(this,"Đã lưu lớp học",Toast.LENGTH_SHORT).show();
            if(!isEditMode){
                android.content.Intent intent=new android.content.Intent(this,ClassDetailActivity.class);
                intent.putExtra(ClassDetailActivity.EXTRA_CLASS_ID,c.getId());startActivity(intent);
            }
            finish();
        });
    }

    private void checkDirtyAndExit() {
        checkIfDirty();
        if (isFormDirty) {
            new MaterialAlertDialogBuilder(this)
                    .setTitle("Rời form có thay đổi?")
                    .setMessage("Bạn có các thay đổi chưa được lưu. Bạn có chắc chắn muốn thoát?")
                    .setPositiveButton("Rời khỏi", (dialog, which) -> finish())
                    .setNegativeButton("Ở lại", (dialog, which) -> dialog.dismiss())
                    .show();
        } else {
            finish();
        }
    }
    @Override protected void onSaveInstanceState(@androidx.annotation.NonNull Bundle state){state.putString("creation_id",creationId);super.onSaveInstanceState(state);}
}

    private void filterAndSortStudents(String query) {
        List<Student> list = AttendanceRepository.getInstance().getStudentsForClass(classId);
        if (list == null || list.isEmpty()) {
            findViewById(R.id.tv_student_search_empty).setVisibility(View.GONE);
            llEmpty.setVisibility(View.VISIBLE);
            rvStudents.setVisibility(View.GONE);
            return;
        }

        llEmpty.setVisibility(View.GONE);
        rvStudents.setVisibility(View.VISIBLE);

        List<Student> filtered = new ArrayList<>();
        String q = query != null ? query.trim().toLowerCase() : "";

        for (Student s : list) {
            if (q.isEmpty()
                    || s.getFullName().toLowerCase().contains(q)
                    || s.getMssv().toLowerCase().contains(q)
                    || (s.getCardBarcode() != null && s.getCardBarcode().toLowerCase().contains(q))) {
                filtered.add(s);
            }
        }

        // Apply Sorting
        if (sortByName) {
            Collections.sort(filtered, (a, b) -> a.getFullName().compareToIgnoreCase(b.getFullName()));
        } else {
            Collections.sort(filtered, (a, b) -> a.getMssv().compareToIgnoreCase(b.getMssv()));
        }

        adapter.setStudents(filtered);
        findViewById(R.id.tv_student_search_empty).setVisibility(filtered.isEmpty()?View.VISIBLE:View.GONE);
    }

public class TeacherAccountsActivity extends DataActivity {
    private RecyclerView list;
    private TeacherAdapter adapter;
    private EditText search;
    private TextView results,summary;
    private TeacherDirectory.Sort sort=TeacherDirectory.Sort.NAME_ASC;
    private String[] sortLabels;
    @Override protected void onReady(Bundle state){
        setContentView(R.layout.activity_teacher_accounts);
        Toolbar bar=findViewById(R.id.toolbar_accounts);setSupportActionBar(bar);
        getSupportActionBar().setTitle(R.string.teachers_title);getSupportActionBar().setDisplayHomeAsUpEnabled(true);bar.setNavigationOnClickListener(v->finish());

        sortLabels=getResources().getStringArray(R.array.teachers_sort_labels);
        if(state!=null){try{sort=TeacherDirectory.Sort.valueOf(state.getString("teacher_sort","NAME_ASC"));}catch(IllegalArgumentException ignored){}search.setText(state.getString("teacher_query",""));}
        list=findViewById(R.id.accounts_list);list.setLayoutManager(new LinearLayoutManager(this));list.setHasFixedSize(true);
        adapter=new TeacherAdapter(new TeacherAdapter.Actions(){
            @Override public void assign(Account user){TeacherAccountsActivity.this.assign(user);}
            @Override public void edit(Account user){editor(user);}

        findViewById(R.id.accounts_add).setOnClickListener(v->editor(null));

        findViewById(R.id.accounts_clear_search).setOnClickListener(v->search.setText(""));
        search.setOnEditorActionListener((v,action,event)->{
            if(action!=android.view.inputmethod.EditorInfo.IME_ACTION_SEARCH)return false;
            android.view.inputmethod.InputMethodManager keyboard=(android.view.inputmethod.InputMethodManager)getSystemService(INPUT_METHOD_SERVICE);
            if(keyboard!=null)keyboard.hideSoftInputFromWindow(search.getWindowToken(),0);search.clearFocus();return true;
        });
        search.addTextChangedListener(new TextWatcher(){
            public void beforeTextChanged(CharSequence s,int start,int count,int after){}
            public void onTextChanged(CharSequence s,int start,int before,int count){render();list.scrollToPosition(0);}
            public void afterTextChanged(Editable value){}
        });
    }
    @Override protected void onDataResume(){render();}
    @Override protected void onSaveInstanceState(android.os.Bundle state){
        if(search!=null){state.putString("teacher_query",search.getText().toString());state.putString("teacher_sort",sort.name());}
        super.onSaveInstanceState(state);
    }
    private void render(){
        AttendanceRepository repo=AttendanceRepository.getInstance();List<Account> teachers=repo.getTeachers();Map<String,Integer> counts=new HashMap<>();int active=0;
        for(Account user:teachers){counts.put(user.id,repo.assignedClassIds(user.id).size());if(user.enabled)active++;}
        List<TeacherDirectory.Entry> rows=TeacherDirectory.select(teachers,counts,search.getText().toString(),sort);adapter.setRows(rows);
        summary.setText(getString(R.string.teachers_summary,active,teachers.size()-active));
        results.setText(getString(R.string.teachers_results,rows.size(),teachers.size()));
        ((TextView)findViewById(R.id.accounts_sort)).setText(sortLabels[sort.ordinal()]);
        boolean empty=rows.isEmpty(),noAccounts=teachers.isEmpty();
        list.setVisibility(empty?View.GONE:View.VISIBLE);findViewById(R.id.accounts_empty).setVisibility(empty?View.VISIBLE:View.GONE);
        ((TextView)findViewById(R.id.accounts_empty_title)).setText(noAccounts?R.string.teachers_empty:R.string.teachers_no_results);
        ((TextView)findViewById(R.id.accounts_empty_description)).setText(noAccounts?R.string.teachers_empty_description:R.string.teachers_no_results_description);
        findViewById(R.id.accounts_clear_search).setVisibility(search.getText().toString().trim().isEmpty()?View.GONE:View.VISIBLE);
    }
    private void sortDialog(){
        new MaterialAlertDialogBuilder(this).setTitle(R.string.teachers_sort)
            .setSingleChoiceItems(sortLabels,sort.ordinal(),(dialog,index)->{sort=TeacherDirectory.Sort.values()[index];render();list.scrollToPosition(0);dialog.dismiss();})
            .setNegativeButton("Đóng",null).show();
    }
    private void toggleAccount(Account user){
        new MaterialAlertDialogBuilder(this).setTitle(user.enabled?"Khóa tài khoản?":"Mở khóa tài khoản?")
            .setMessage(user.fullName+"\n"+user.username+"\n\nGiữ nguyên phân công và lịch sử điểm danh.")
            .setPositiveButton("Xác nhận",(d,w)->Tasks.run(this,null,()->{AttendanceRepository.getInstance().updateTeacher(user.id,user.fullName,!user.enabled,null);return true;},ok->render()))
            .setNegativeButton("Hủy",null).show();
    }
    private void editor(Account user){
        View form=getLayoutInflater().inflate(R.layout.dialog_account,null);EditText username=form.findViewById(R.id.account_username),name=form.findViewById(R.id.account_name),password=form.findViewById(R.id.account_password);TextView error=form.findViewById(R.id.account_error);
        if(user!=null){username.setText(user.username);username.setEnabled(false);name.setText(user.fullName);((com.google.android.material.textfield.TextInputLayout)form.findViewById(R.id.account_password_layout)).setHint("Mật khẩu mới (để trống nếu giữ nguyên)");}
        AlertDialog dialog=new MaterialAlertDialogBuilder(this).setTitle(user==null?"Thêm giảng viên":"Cập nhật giảng viên").setView(form).setPositiveButton("Lưu",null).setNegativeButton("Hủy",null).create();
        dialog.setOnShowListener(d->dialog.getButton(-1).setOnClickListener(v->{String login=username.getText().toString(),full=name.getText().toString(),secret=password.getText().toString();Tasks.run(this,v,()->{if(user==null)AttendanceRepository.getInstance().createTeacher(login,full,secret);else AttendanceRepository.getInstance().updateTeacher(user.id,full,user.enabled,secret);return true;},ok->{password.setText("");dialog.dismiss();if(user==null)search.setText("");render();Toast.makeText(this,user==null?"Đã thêm giảng viên":"Đã cập nhật tài khoản",Toast.LENGTH_SHORT).show();},error::setText);}));dialog.show();
    }
    private void assign(Account user){
        AttendanceRepository repo=AttendanceRepository.getInstance();List<ClassRoom> classes=new ArrayList<>();
        for(ClassRoom classroom:repo.getAllClasses()){
            List<String> assigned=repo.assignedTeacherIds(classroom.getId());
            if(assigned.isEmpty()||(assigned.size()==1&&assigned.contains(user.id)))classes.add(classroom);
        }
        if(classes.isEmpty()){new MaterialAlertDialogBuilder(this).setMessage("Không có lớp chưa phân công. Để đổi giảng viên, vào Chi tiết lớp → Phân công giảng viên.").setPositiveButton("Đóng",null).show();return;}
        List<String> current=repo.assignedClassIds(user.id);boolean[] selected=new boolean[classes.size()];String[] labels=new String[classes.size()];for(int i=0;i<classes.size();i++){selected[i]=current.contains(classes.get(i).getId());labels[i]=classes.get(i).getName()+" • "+classes.get(i).getCode();}
        AlertDialog dialog=new MaterialAlertDialogBuilder(this).setTitle("Phân công: "+user.fullName).setMultiChoiceItems(labels,selected,(d,n,checked)->selected[n]=checked).setPositiveButton("Lưu phân công",null).setNegativeButton("Hủy",null).create();
        dialog.setOnShowListener(d->dialog.getButton(-1).setOnClickListener(v->{List<String> ids=new ArrayList<>();for(int i=0;i<selected.length;i++)if(selected[i])ids.add(classes.get(i).getId());Tasks.run(this,v,()->{repo.replaceAssignments(user.id,ids);return true;},ok->{dialog.dismiss();render();});}));dialog.show();
    }
}

package com.example.th9.ui;

import android.Manifest;
import android.content.Intent;
import android.content.pm.PackageManager;
import android.net.Uri;
import android.os.*;
import android.provider.Settings;
import android.view.View;
import android.widget.*;
import androidx.activity.result.ActivityResultLauncher;
import androidx.activity.result.contract.ActivityResultContracts;
import androidx.camera.core.Camera;
import androidx.camera.view.PreviewView;
import androidx.core.content.ContextCompat;
import com.example.th9.R;
import com.example.th9.camera.CameraScanner;
import com.example.th9.data.*;
import com.example.th9.model.*;
import com.google.android.material.dialog.MaterialAlertDialogBuilder;

public class ScanActivity extends DataActivity {
    public static final String EXTRA_MODE="extra_mode",EXTRA_SESSION_ID="extra_session_id";
    public static final int MODE_ATTENDANCE=1,MODE_VERIFY_CARD=3;
    private int mode;
    private String sessionId;
    private CameraScanner scanner;
    private Camera camera;
    private boolean modal,busy,torch,uiReady,activityResumed;
    private String lastCode;
    private long lastCodeAt;
    private TextView title,subtitle,status,info,time,error;
    private View errorOverlay,reticle;
    private ImageButton flash;
    private final ActivityResultLauncher<String> permission=registerForActivityResult(new ActivityResultContracts.RequestPermission(),granted->{
        if(granted)startCamera();else cameraError("Bạn chưa cấp quyền camera");
    });
    @Override protected void onReady(Bundle state){
        setContentView(R.layout.activity_scan);
        mode=getIntent().getIntExtra(EXTRA_MODE,MODE_ATTENDANCE)==MODE_ATTENDANCE?MODE_ATTENDANCE:MODE_VERIFY_CARD;sessionId=getIntent().getStringExtra(EXTRA_SESSION_ID);
        title=findViewById(R.id.tv_scan_header_title);subtitle=findViewById(R.id.tv_scan_header_subtitle);
        status=findViewById(R.id.tv_result_status_title);info=findViewById(R.id.tv_result_student_info);time=findViewById(R.id.tv_result_time_badge);
        error=findViewById(R.id.tv_camera_error);errorOverlay=findViewById(R.id.ll_camera_error_overlay);reticle=findViewById(R.id.fl_reticle_frame);
        flash=findViewById(R.id.btn_scan_flash);flash.setEnabled(false);
        findViewById(R.id.btn_scan_back).setOnClickListener(v->finish());
        findViewById(R.id.ll_scan_actions_attendance).setVisibility(mode==MODE_ATTENDANCE?View.VISIBLE:View.GONE);
        findViewById(R.id.btn_fallback_manual).setVisibility(mode==MODE_ATTENDANCE?View.VISIBLE:View.GONE);
        findViewById(R.id.btn_scan_manual_input).setOnClickListener(v->manual());
        findViewById(R.id.btn_fallback_manual).setOnClickListener(v->manual());
        findViewById(R.id.btn_finish_scan).setOnClickListener(v->finishSession());
        findViewById(R.id.btn_camera_retry).setOnClickListener(v->{
            boolean asked=getSharedPreferences("settings",MODE_PRIVATE).getBoolean("camera_asked",false);
            if(ContextCompat.checkSelfPermission(this,Manifest.permission.CAMERA)!=PackageManager.PERMISSION_GRANTED && asked && !shouldShowRequestPermissionRationale(Manifest.permission.CAMERA))
                startActivity(new Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS,Uri.parse("package:"+getPackageName())));
            else requestCamera();
        });
        flash.setOnClickListener(v->{if(camera!=null&&camera.getCameraInfo().hasFlashUnit()){
            torch=!torch;camera.getCameraControl().enableTorch(torch);
            flash.setImageResource(torch?R.drawable.ic_flash_on:R.drawable.ic_flash_off);
        }});
        getSupportFragmentManager().setFragmentResultListener("attendance_changed",this,(key,result)->refresh());
        getSupportFragmentManager().registerFragmentLifecycleCallbacks(new androidx.fragment.app.FragmentManager.FragmentLifecycleCallbacks(){
            @Override public void onFragmentDetached(androidx.fragment.app.FragmentManager manager,androidx.fragment.app.Fragment fragment){
                if(fragment instanceof ManualAttendanceBottomSheetDialog){modal=false;pauseScanner();refresh();}
            }
        },false);
        modal=getSupportFragmentManager().findFragmentByTag("manual_attendance")!=null;
        uiReady=true;refresh();requestCamera();
    }
    private void requestCamera(){
        if(ContextCompat.checkSelfPermission(this,Manifest.permission.CAMERA)==PackageManager.PERMISSION_GRANTED)startCamera();
        else {getSharedPreferences("settings",MODE_PRIVATE).edit().putBoolean("camera_asked",true).apply();permission.launch(Manifest.permission.CAMERA);}
    }
    private void startCamera(){
        if(!uiReady||isFinishing())return;
        if(scanner!=null)scanner.close();
        camera=null;torch=false;flash.setImageResource(R.drawable.ic_flash_off);
        errorOverlay.setVisibility(View.GONE);reticle.setVisibility(View.VISIBLE);flash.setEnabled(false);
        PreviewView preview=findViewById(R.id.camera_preview);
        scanner=new CameraScanner(this,this,preview,new CameraScanner.Listener(){
            @Override public void onCode(String raw,String format){handleCode(raw,format);}
            @Override public void onMessage(String message){ /* Keep decoder guidance out of the scan interface. */ }
            @Override public void onCamera(Camera value){camera=value;flash.setEnabled(value.getCameraInfo().hasFlashUnit());}
            @Override public void onError(String message){cameraError(message);}
        });
        pauseScanner();scanner.start();
    }
    private void cameraError(String message){
        if(!uiReady)return;error.setText(message);errorOverlay.setVisibility(View.VISIBLE);reticle.setVisibility(View.GONE);flash.setEnabled(false);
        ((TextView)findViewById(R.id.btn_camera_retry)).setText(message.startsWith("Bạn chưa cấp quyền")&&!shouldShowRequestPermissionRationale(Manifest.permission.CAMERA)?"Mở cài đặt ứng dụng":"Thử lại");
    }
    private void handleCode(String raw,String format){
        if(modal||busy)return;
        long elapsed=SystemClock.elapsedRealtime();if(raw.equals(lastCode)&&elapsed-lastCodeAt<2500)return;
        lastCode=raw;lastCodeAt=elapsed;
        if(mode==MODE_VERIFY_CARD){
            busy=true;pauseScanner();
            Tasks.run(this,null,()->AttendanceRepository.getInstance().verifyCard(raw,format),student->{
                busy=false;pauseScanner();show("Thẻ trùng khớp",student.getFullName()+" · MSSV: "+student.getMssv()+"\nNgày sinh: "+(student.getBirthDate()==null?"—":student.getBirthDate()),"Chỉ tra cứu thông tin thẻ");
            },message->{busy=false;pauseScanner();show(message,raw,"");});return;
        }
        busy=true;pauseScanner();
        Tasks.run(this,null,()->AttendanceRepository.getInstance().scan(sessionId,raw,format,System.currentTimeMillis()),record->{
            busy=false;pauseScanner();show("Ghi nhận thành công",record.getNameSnapshot()+" · "+record.getStatusDisplayName(),record.getCheckInTime());feedback();refresh();
        },message->{busy=false;pauseScanner();show(message,raw,"");refresh();});
    }
    private void show(String result,String details,String recorded){
        status.setText(result);info.setText(details);info.setVisibility(View.VISIBLE);time.setText(recorded);
        ((ImageView)findViewById(R.id.iv_result_icon)).setImageResource((result.startsWith("Ghi nhận thành công")||result.equals("Thẻ trùng khớp"))?R.drawable.ic_check_circle:result.startsWith("Đã đọc")?R.drawable.ic_barcode:R.drawable.ic_cancel);
        View card=findViewById(R.id.card_scan_result);card.animate().cancel();card.setAlpha(.4f);card.animate().alpha(1f).setDuration(220).start();
    }
    private void refresh(){
        if(mode==MODE_VERIFY_CARD){title.setText("Quét kiểm tra thẻ");subtitle.setText("Tra cứu hồ sơ · Không ghi điểm danh");return;}
        AttendanceSession s=AttendanceRepository.getInstance().getSessionById(sessionId);
        if(s==null||!AttendanceRepository.getInstance().canControlSession(sessionId)){finish();return;}
        if(s.getStatus()==AttendanceSession.SessionStatus.CLOSED){resultScreen();return;}
        if(s.getStatus()!=AttendanceSession.SessionStatus.OPEN){finish();return;}
        title.setText(s.getTitle());int checked=AttendanceRepository.getInstance().getCheckedInTotal(sessionId);
        subtitle.setText(AttendanceRepository.getInstance().sessionClassLabel(s)+" · "+checked+"/"+s.getExpectedCount());
        subtitle.animate().cancel();subtitle.setAlpha(.5f);subtitle.animate().alpha(1).setDuration(250).start();
    }
    private void manual(){
        if(modal||busy||mode!=MODE_ATTENDANCE)return;modal=true;pauseScanner();
        ManualAttendanceBottomSheetDialog.newInstance(sessionId,this::refresh).show(getSupportFragmentManager(),"manual_attendance");
    }
    private void finishSession(){
        if(modal||busy)return;modal=true;pauseScanner();int pending=AttendanceRepository.getInstance().getUncheckedCount(sessionId);
        androidx.appcompat.app.AlertDialog dialog=new MaterialAlertDialogBuilder(this).setTitle("Kết thúc buổi?")
            .setMessage("Còn "+pending+" sinh viên chưa điểm danh. Những sinh viên này sẽ được ghi Vắng.")
            .setPositiveButton("Kết thúc",(d,w)->{busy=true;Tasks.run(this,findViewById(R.id.btn_finish_scan),()->{
                AttendanceRepository.getInstance().finishSession(sessionId);return true;
            },ok->resultScreen(),message->{busy=false;pauseScanner();show(message,"","");});})
            .setNegativeButton("Tiếp tục",null).create();
        dialog.setOnDismissListener(d->{modal=false;pauseScanner();});dialog.show();
    }
    private void resultScreen(){Intent i=new Intent(this,SessionResultActivity.class);i.putExtra(SessionResultActivity.EXTRA_SESSION_ID,sessionId);startActivity(i);finish();}
    // DataActivity calls onDataResume from inside onResume, before AndroidX necessarily
    // dispatches ON_RESUME. Using the registry there can leave analysis paused forever
    // even though CameraX later starts its preview. Resume explicitly in onPostResume.
    private void pauseScanner(){if(scanner!=null)scanner.setPaused(modal||busy||!activityResumed);}
    private void feedback(){
        android.content.SharedPreferences p=getSharedPreferences("settings",MODE_PRIVATE);
        if(p.getBoolean("sound",true)){
            android.media.ToneGenerator tone=new android.media.ToneGenerator(android.media.AudioManager.STREAM_NOTIFICATION,70);
            tone.startTone(android.media.ToneGenerator.TONE_PROP_BEEP,120);new Handler(Looper.getMainLooper()).postDelayed(tone::release,200);
        }
        if(p.getBoolean("vibrate",true)){
            Vibrator vibrator=(Vibrator)getSystemService(VIBRATOR_SERVICE);
            if(vibrator!=null&&vibrator.hasVibrator()){if(Build.VERSION.SDK_INT>=26)vibrator.vibrate(VibrationEffect.createOneShot(80,VibrationEffect.DEFAULT_AMPLITUDE));else vibrator.vibrate(80);}
        }
    }
    @Override protected void onDataResume(){if(uiReady){refresh();pauseScanner();if(scanner==null&&ContextCompat.checkSelfPermission(this,Manifest.permission.CAMERA)==PackageManager.PERMISSION_GRANTED)startCamera();}}
    @Override protected void onPostResume(){super.onPostResume();activityResumed=true;pauseScanner();}
    @Override protected void onPause(){activityResumed=false;pauseScanner();super.onPause();}
    @Override protected void onDestroy(){if(scanner!=null)scanner.close();super.onDestroy();}
}

    public static final String EXTRA_PRESELECTED_CLASS_ID = "extra_preselected_class_id";
    public static final String EXTRA_SESSION_ID = "edit_session_id";
    private String editSessionId;
    private String creationId;

    private String preselectedClassId;
    private String selectedClassId;

    private Toolbar toolbar;
    private TextInputLayout tilClass, tilTitle, tilDate, tilTime, tilLateTime;
    private AutoCompleteTextView actvClass;
    private TextInputEditText etTitle, etDate, etTime, etLateTime;
    private MaterialSwitch switchLate;
    private LinearLayout llLateContainer;
    private TextView tvExpectedCount;
    private MaterialButton btnSubmit, btnPreview, btnCancel;

    private final Calendar calendar = Calendar.getInstance();
    private List<ClassRoom> classesList = new ArrayList<>();

    @Override
    protected void onReady(Bundle savedInstanceState) {

        setContentView(R.layout.activity_session_form);

        preselectedClassId = getIntent().getStringExtra(EXTRA_PRESELECTED_CLASS_ID);
        creationId=savedInstanceState==null?AttendanceRepository.id():savedInstanceState.getString("creation_id",AttendanceRepository.id());

        initViews();
        setupToolbar();
        setupClassesDropdown();
        setupDateTimePickers();
        setupLateToggle();
        setupListeners();
        editSessionId=getIntent().getStringExtra(EXTRA_SESSION_ID);
        if(editSessionId!=null){
            AttendanceSession s=AttendanceRepository.getInstance().getSessionById(editSessionId);
            if(s==null||!AttendanceRepository.getInstance().canControlSession(editSessionId)||s.getStatus()!=AttendanceSession.SessionStatus.DRAFT){finish();return;}
            selectedClassId=s.getClassId();
            calendar.setTimeInMillis(s.getScheduledAt());
            updateExpectedCount(AttendanceRepository.getInstance().getClassById(selectedClassId));
            for(int i=0;i<classesList.size();i++)if(classesList.get(i).getId().equals(selectedClassId))
                actvClass.setText(classesList.get(i).getName()+" ("+classesList.get(i).getCode()+")",false);
            etTitle.setText(s.getTitle());etDate.setText(s.getDate());etTime.setText(s.getStartTime());
            switchLate.setChecked(s.isEnableLateThreshold());etLateTime.setText(s.getLateThresholdTime());btnSubmit.setText("Lưu thay đổi");
        }
        if(savedInstanceState!=null){selectedClassId=savedInstanceState.getString("selectedClassId",selectedClassId);updateExpectedCount(AttendanceRepository.getInstance().getClassById(selectedClassId));}
        if(classesList.isEmpty()){
            btnSubmit.setEnabled(false);tvExpectedCount.setText(AttendanceRepository.getInstance().isAdmin()?"Chưa có lớp. Hãy tạo lớp trước.":"Chưa được phân công lớp. Liên hệ quản trị viên.");
            btnPreview.setVisibility(AttendanceRepository.getInstance().isAdmin()?View.VISIBLE:View.GONE);btnPreview.setText("Tạo lớp");
            btnPreview.setOnClickListener(v->{startActivity(new Intent(this,ClassFormActivity.class));finish();});
        }
    }

    private void initViews() {
        toolbar = findViewById(R.id.toolbar_session_form);
        tilClass = findViewById(R.id.til_session_class);
        tilTitle = findViewById(R.id.til_session_title);
        tilDate = findViewById(R.id.til_session_date);
        tilTime = findViewById(R.id.til_session_time);
        tilLateTime = findViewById(R.id.til_session_late_time);

        actvClass = findViewById(R.id.actv_session_class);

        llLateContainer = findViewById(R.id.ll_late_threshold_container);

        btnSubmit = findViewById(R.id.btn_session_submit);
        btnPreview = findViewById(R.id.btn_session_preview);
        btnCancel = findViewById(R.id.btn_session_cancel);
    }

    private void setupToolbar() {
        setSupportActionBar(toolbar);
        if (getSupportActionBar() != null) {
            getSupportActionBar().setDisplayHomeAsUpEnabled(true);
            getSupportActionBar().setTitle(R.string.session_form_create_title);
        }
        toolbar.setNavigationOnClickListener(v -> finish());
    }

    private void setupClassesDropdown() {
        classesList = AttendanceRepository.getInstance().getAllClasses();
        List<String> displayNames = new ArrayList<>();
        int selectedIndex = 0;

        for (int i = 0; i < classesList.size(); i++) {
            ClassRoom c = classesList.get(i);
            displayNames.add(c.getName() + " (" + c.getCode() + ")");
            if (preselectedClassId != null && preselectedClassId.equals(c.getId())) {
                selectedIndex = i;
            }
        }

        ArrayAdapter<String> adapter = new ArrayAdapter<>(this, android.R.layout.simple_dropdown_item_1line, displayNames);
        actvClass.setAdapter(adapter);

        if (!classesList.isEmpty()) {
            actvClass.setText(displayNames.get(selectedIndex), false);
            selectedClassId = classesList.get(selectedIndex).getId();
            updateExpectedCount(classesList.get(selectedIndex));
        }

        actvClass.setOnItemClickListener((parent, view, position, id) -> {
            ClassRoom selected = classesList.get(position);
            selectedClassId = selected.getId();
            updateExpectedCount(selected);
            tilClass.setError(null);
        });
    }

    private void updateExpectedCount(ClassRoom c) {
        if (c != null) {
            tvExpectedCount.setText("Sĩ số dự kiến: " + c.getStudentCount() + " sinh viên");
            if (etTitle.getText() == null || etTitle.getText().toString().isEmpty()) {
                etTitle.setText("Điểm danh thường kỳ");
            }
        }

    private void validateAndCreateSession() {
        String title = etTitle.getText() != null ? etTitle.getText().toString().trim() : "";
        String date = etDate.getText() != null ? etDate.getText().toString().trim() : "";
        String time = etTime.getText() != null ? etTime.getText().toString().trim() : "";
        boolean enableLate = switchLate.isChecked();
        String lateTime = etLateTime.getText() != null ? etLateTime.getText().toString().trim() : "";

        boolean hasError = false;
        if (selectedClassId == null) {
            tilClass.setError("Vui lòng chọn lớp học");
            hasError = true;
        } else {
            tilClass.setError(null);
        }

        if (title.isEmpty()) {
            tilTitle.setError("Vui lòng nhập tên buổi");
            hasError = true;
        } else {
            tilTitle.setError(null);
        }

        if (hasError) return;

        ClassRoom cr = AttendanceRepository.getInstance().getClassById(selectedClassId);
        int expected = cr != null ? cr.getStudentCount() : 0;

        if(editSessionId!=null){createAndOpen(title,date,time,enableLate,lateTime,expected,AttendanceSession.SessionStatus.DRAFT);return;}
        new MaterialAlertDialogBuilder(this).setTitle("Tạo buổi ngoài lịch?")

            .setPositiveButton("Tạo ngoài lịch",(d,w)->createAndOpen(title,date,time,enableLate,lateTime,expected,AttendanceSession.SessionStatus.DRAFT))
            .setNegativeButton("Hủy",null).show();
    }

    private void createAndOpen(String title, String date, String time, boolean enableLate, String lateTime, int expected, AttendanceSession.SessionStatus status) {
        AttendanceSession session = new AttendanceSession(
                null, selectedClassId, title, date, time, enableLate, lateTime,
                status, null, expected
        );
        session.setId(editSessionId==null?creationId:editSessionId);
        com.example.th9.data.Tasks.run(this,btnSubmit,()->{
            if(editSessionId==null)AttendanceRepository.getInstance().addSession(session);

            return session;
        },saved->{
            Toast.makeText(this,"Đã lưu buổi chưa mở",Toast.LENGTH_SHORT).show();
            Intent intent=new Intent(this,SessionDetailActivity.class);
            intent.putExtra(SessionDetailActivity.EXTRA_SESSION_ID,saved.getId());startActivity(intent);finish();
        });
    }

    private void loadSessionData() {
        session = AttendanceRepository.getInstance().getSessionById(sessionId);
        if (session == null) {finish();return;}

        boolean control=!AttendanceRepository.getInstance().isAdmin()&&AttendanceRepository.getInstance().canControlSession(sessionId);
        for(View button:new View[]{btnOpen,btnScan,btnManual,btnFinish})button.setVisibility(control?View.VISIBLE:View.GONE);
        tvTitle.setText(session.getTitle());
        ClassRoom cr = AttendanceRepository.getInstance().getClassById(session.getClassId());
        if (cr != null) {
            tvClass.setText(AttendanceRepository.getInstance().sessionClassLabel(session));
        }
        tvSchedule.setText(session.getStartTime() + " • " + session.getDate()+"\n"+session.getScheduleLabel()+"\nGiảng viên: "+session.getTeacherNameSnapshot());
        tvExpected.setText("Sĩ số: " + session.getExpectedCount() + " sinh viên");

        if (session.isEnableLateThreshold() && session.getLateThresholdTime() != null) {
            tvLateRule.setText("Quy tắc: Sau " + session.getLateThresholdTime() + " tính đi muộn");
            tvLateRule.setVisibility(View.VISIBLE);
        } else {
            tvLateRule.setVisibility(View.GONE);
        }

        if (session.getFinishedTime() != null) {
            tvFinishedTime.setText("Kết thúc lúc: " + session.getFinishedTime());
            tvFinishedTime.setVisibility(View.VISIBLE);
        } else {
            tvFinishedTime.setVisibility(View.GONE);
        }

        // Toggle state containers
        switch (session.getStatus()) {
            case DRAFT:
                layoutUpcoming.setVisibility(View.VISIBLE);
                layoutActive.setVisibility(View.GONE);
                layoutFinished.setVisibility(View.GONE);

                tvBadge.setText(R.string.session_status_upcoming);
                tvBadge.setBackgroundResource(R.drawable.bg_badge_unchecked);
                tvBadge.setTextColor(ContextCompat.getColor(this, R.color.text_secondary));
                break;

            case OPEN:
                layoutUpcoming.setVisibility(View.GONE);
                layoutActive.setVisibility(View.VISIBLE);
                layoutFinished.setVisibility(View.GONE);

                tvBadge.setText(R.string.session_status_active);
                tvBadge.setBackgroundResource(R.drawable.bg_badge_present);
                tvBadge.setTextColor(ContextCompat.getColor(this, R.color.on_status_present_container));

                int present = AttendanceRepository.getInstance().getPresentCount(sessionId);
                int late = AttendanceRepository.getInstance().getLateCount(sessionId);
                int unchecked = AttendanceRepository.getInstance().getUncheckedCount(sessionId);
                int checkedIn = present + late;
                int total = session.getExpectedCount();
                int pct = total > 0 ? (checkedIn * 100 / total) : 0;

                tvActiveRatio.setText(checkedIn + "/" + total + " đã điểm danh (" + pct + "%)");
                progressActive.setProgress(pct);

                tvActivePresent.setText(present + " Có mặt");
                tvActiveLate.setText(late + " Đi muộn");
                tvActiveUnchecked.setText(unchecked + " Chưa điểm danh");
                break;

            case CLOSED:
                layoutUpcoming.setVisibility(View.GONE);
                layoutActive.setVisibility(View.GONE);
                layoutFinished.setVisibility(View.VISIBLE);

                tvBadge.setText(R.string.session_status_finished);
                tvBadge.setBackgroundResource(R.drawable.bg_badge_late);
                tvBadge.setTextColor(ContextCompat.getColor(this, R.color.on_status_late_container));

                int fPresent = AttendanceRepository.getInstance().getPresentCount(sessionId);
                int fLate = AttendanceRepository.getInstance().getLateCount(sessionId);
                int fAbsent = AttendanceRepository.getInstance().getAbsentCount(sessionId);
                int fTotal = session.getExpectedCount();
                int fPct = fTotal > 0 ? ((fPresent + fLate) * 100 / fTotal) : 0;

                tvFinishedSummary.setText("• Có mặt: " + fPresent + "  • Đi muộn: " + fLate + "  • Vắng mặt: " + fAbsent + "\n• Tỷ lệ chuyên cần: " + fPct + "%");
                break;
        }

        classes.setOnItemSelectedListener(new Selection(()->{int p=classes.getSelectedItemPosition();classId=p>0&&p<=classOptions.size()?classOptions.get(p-1).getId():null;render();}));
        teachers.setOnItemSelectedListener(new Selection(()->{int p=teachers.getSelectedItemPosition();teacherId=p>0&&p<=teacherOptions.size()?teacherOptions.get(p-1).id:null;render();}));
    }
    private class Selection implements AdapterView.OnItemSelectedListener {
        final Runnable changed;Selection(Runnable changed){this.changed=changed;}
        public void onItemSelected(AdapterView<?> p,View v,int position,long id){if(!binding)changed.run();}
        public void onNothingSelected(AdapterView<?> p){}
    }
    @Override protected void onDataResume(){
        binding=true;classOptions=repo.getAllClasses();List<String> labels=new ArrayList<>();labels.add("Tất cả lớp");int selected=0;
        for(ClassRoom c:classOptions){labels.add(c.getName()+" • "+c.getCode());if(c.getId().equals(classId))selected=labels.size()-1;}
        classes.setAdapter(new ArrayAdapter<>(this,android.R.layout.simple_spinner_dropdown_item,labels));classes.setSelection(selected);
        if(selected==0)classId=null;
        teacherOptions=repo.isAdmin()?repo.getTeachers():Collections.emptyList();labels=new ArrayList<>();labels.add("Tất cả giảng viên");selected=0;
        for(Account t:teacherOptions){labels.add(t.fullName+" ("+t.username+")");if(t.id.equals(teacherId))selected=labels.size()-1;}
        teachers.setAdapter(new ArrayAdapter<>(this,android.R.layout.simple_spinner_dropdown_item,labels));teachers.setSelection(selected);
        if(selected==0)teacherId=null;binding=false;render();
    }
    private void render(){
        if(adapter==null)return;
        findViewById(R.id.week_navigation).setVisibility(manage?View.GONE:View.VISIBLE);
        ((TextView)findViewById(R.id.week_label)).setText(ScheduleRules.display(ScheduleRules.iso(week))+" – "+ScheduleRules.display(ScheduleRules.iso(week+6)));
        List<Entry> entries=new ArrayList<>();List<String> allowed=teacherId==null?null:repo.assignedClassIds(teacherId);
        for(ClassSchedule rule:repo.getSchedules()){
            if(classId!=null&&!classId.equals(rule.classId)||allowed!=null&&!allowed.contains(rule.classId))continue;
            if(manage)entries.add(new Entry(rule,null));
            else for(long d=week;d<week+7;d++)if(ScheduleRules.occurs(rule,d))entries.add(new Entry(rule,ScheduleRules.iso(d)));
        }
        entries.sort(Comparator.comparing((Entry e)->e.date==null?e.rule.startDate:e.date).thenComparingInt(e->e.rule.startMinute).thenComparing(e->e.rule.classId));
        adapter.entries=entries;adapter.notifyDataSetChanged();findViewById(R.id.schedule_empty).setVisibility(entries.isEmpty()?View.VISIBLE:View.GONE);
        RecyclerView list=findViewById(R.id.schedule_list);list.setAlpha(0.4f);list.animate().alpha(1f).setDuration(160).start();
    }

            h.info.setText(range+"\n"+ScheduleRules.shift(s)+" • "+ScheduleRules.time(s.startMinute)+"–"+ScheduleRules.time(s.endMinute));

                if(cmp==0){bg=R.color.primary_container;fg=R.color.on_primary_container;}

        }
